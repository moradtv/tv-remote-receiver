    <div>
        <button class="btn" onclick="sendCommand('netflix')">تشغيل Netflix</button>
        <button class="btn" onclick="sendCommand('youtube')">تشغيل YouTube</button>
        <button class="btn" style="background: #f44336;" onclick="sendCommand('off')">إطفاء الشاشة</button>
    </div>

    <h3 id="feedback" style="margin-top: 30px; color: #ffeb3b;"></h3>

    <script>
        // 1. معلومات الربط الخاصة بـ Firebase (عوضهم بالمعلومات الحقيقية ديال مشروعك)
        const firebaseConfig = {
            apiKey: "YOUR_API_KEY",
            authDomain: "YOUR_AUTH_DOMAIN",
            databaseURL: "YOUR_DATABASE_URL",
            projectId: "YOUR_PROJECT_ID",
            storageBucket: "YOUR_STORAGE_BUCKET",
            messagingSenderId: "YOUR_MESSAGING_SENDER_ID",
            appId: "YOUR_APP_ID"
        };

        // تهيئة Firebase
        firebase.initializeApp(firebaseConfig);
        const db = firebase.database();

        // 2. دالة إرسال الأمر إلى Firebase بدون أي أخطاء
        function sendCommand(cmd) {
            db.ref('tv_commands/latest_command').set(cmd)
                .then(() => {
                    document.getElementById('feedback').innerText = "تم إرسال الأمر بنجاح: " + cmd;
                })
                .catch((error) => {
                    document.getElementById('feedback').innerText = "خطأ في الإرسال: " + error.message;
                });
        }
    </script>
